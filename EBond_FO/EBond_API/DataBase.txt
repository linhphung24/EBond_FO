-- ============================================================
-- 1. Add PasswordHash column to MCAccount (if not exists)
-- ============================================================
ALTER TABLE MCAccount
ADD PasswordHash nvarchar(255);

ALTER TABLE MCAccount_Cancel
ADD PasswordHash nvarchar(255);


-- ============================================================
-- 2. Create RefreshTokens table
-- ============================================================
CREATE TABLE RefreshTokens
(
    Id                  UNIQUEIDENTIFIER    PRIMARY KEY DEFAULT NEWID(),
    UserID              INT                 NOT NULL,
    TokenHash           NVARCHAR(255)       NOT NULL,
    ExpiryDate          DATETIME2           NOT NULL,
    CreatedAt           DATETIME2           DEFAULT GETUTCDATE(),
    RevokedAt           DATETIME2           NULL,
    ReplacedByTokenHash NVARCHAR(255)       NULL,
    IpAddress           NVARCHAR(50)        NULL,
    Device              NVARCHAR(100)       NULL,
    UserAgent           NVARCHAR(255)       NULL,

    CONSTRAINT FK_RefreshTokens_MCAccount
        FOREIGN KEY (UserID) REFERENCES MCAccount(Id)
);

CREATE INDEX IX_RefreshTokens_TokenHash ON RefreshTokens(TokenHash);
CREATE INDEX IX_RefreshTokens_UserID    ON RefreshTokens(UserID);


-- ============================================================
-- 3. Stored Procedure: get user by Username
--    Returns: Id, Username, PasswordHash (column order: 0, 1, 2)
-- ============================================================
CREATE OR ALTER PROCEDURE sp_User_GetByUsername
    @Username NVARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;
    SELECT Id, Username, PasswordHash
    FROM   MCAccount
    WHERE  Username = @Username;
END


-- ============================================================
-- 4. Stored Procedure: get user by Id
--    Returns: Id, Username, PasswordHash (column order: 0, 1, 2)
-- ============================================================
CREATE OR ALTER PROCEDURE sp_User_GetById
    @Id INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT Id, Username, PasswordHash
    FROM   MCAccount
    WHERE  Id = @Id;
END
